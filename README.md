# 1. Clonar el repositorio
git clone https://github.com/xRodrigoJrx/InkaPark-MOBILE.git

# 2. Entrar a la carpeta del proyecto
cd InkaPark-MOBILE

# 3. Instalar las dependencias necesarias
npm install

# 4. Iniciar el proyecto con Expo
npx expo start

# 5. Escanear el código QR con Expo Go desde el celular
# También pueden presionar:
# a = abrir en Android
# w = abrir en navegador web
